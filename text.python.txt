import numpy as np
word_to_index = {"cats": 0, "like": 1, "to": 2, "sleep": 3}
index_to_word = {i: w for w, i in word_to_index.items()}
vocab_size = len(word_to_index)

def one_hot(index, size):
    vec = np.zeros(size)
    vec[index] = 1
    return vec

X_words = ["cats", "like", "to"]
Y_word = "sleep"

X = [one_hot(word_to_index[w], vocab_size) for w in X_words]
Y = word_to_index[Y_word]

np.random.seed(42)
hidden_size = 3  
Wxh = np.random.randn(hidden_size, vocab_size) * 0.1
Whh = np.random.randn(hidden_size, hidden_size) * 0.1
Why = np.random.randn(vocab_size, hidden_size) * 0.1

bh = np.zeros((hidden_size, 1))
by = np.zeros((vocab_size, 1))

def relu(x):
    return np.maximum(0, x)

def relu_derivative(x):
    return (x > 0).astype(float)

#  forward 
h = {}
h[-1] = np.zeros((hidden_size, 1))
x = {}

print("==== Forward Pass ====")
for t in range(3):
    x[t] = X[t].reshape(-1, 1)
    h[t] = relu(np.dot(Wxh, x[t]) + np.dot(Whh, h[t-1]) + bh)
    print(f"h[{t}] =\n{h[t]}")

y_linear = np.dot(Why, h[2]) + by
y_pred = np.exp(y_linear) / np.sum(np.exp(y_linear))

loss = -np.log(y_pred[Y][0])
print(f"\nLoss = {loss:.4f}")

#  backward 
dWhy = np.zeros_like(Why)
dWxh = np.zeros_like(Wxh)
dWhh = np.zeros_like(Whh)
dbh = np.zeros_like(bh)
dby = np.zeros_like(by)

dh_next = np.zeros_like(h[0])

dy = y_pred.copy()
dy[Y] -= 1  

dWhy += np.dot(dy, h[2].T)
dby += dy

print("\n==== Backward Pass ====")
for t in reversed(range(3)):
    dh = np.dot(Why.T, dy) + dh_next
    drelu = relu_derivative(h[t]) * dh

    dWxh += np.dot(drelu, x[t].T)
    dWhh += np.dot(drelu, h[t-1].T)
    dbh += drelu

    dh_next = np.dot(Whh.T, drelu)
    print(f"\nStep t={t}")
    print(f"drelu =\n{drelu}")
    print(f"dh_next =\n{dh_next}")

lr = 0.1
for param, dparam in zip([Wxh, Whh, Why, bh, by],
                         [dWxh, dWhh, dWhy, dbh, dby]):
    param -= lr * dparam

print("\n✅ Update complete — the model has now learned one step.")
